-- ParkBay schema (SQLite)
CREATE TABLE IF NOT EXISTS slots (
    id     TEXT PRIMARY KEY,          -- e.g. 'A-03'
    block  TEXT NOT NULL,             -- 'A', 'B'
    number INTEGER NOT NULL
);

CREATE TABLE IF NOT EXISTS bookings (
    id       INTEGER PRIMARY KEY AUTOINCREMENT,
    slot_id  TEXT NOT NULL REFERENCES slots(id),
    name     TEXT NOT NULL,
    vehicle  TEXT NOT NULL,
    hours    INTEGER NOT NULL CHECK (hours BETWEEN 1 AND 12),
    time_in  TEXT NOT NULL,
    time_out TEXT,
    status   TEXT NOT NULL DEFAULT 'active' CHECK (status IN ('active','done'))
);

-- The database itself prevents double booking:
-- one active booking per slot, and one active booking per vehicle.
CREATE UNIQUE INDEX IF NOT EXISTS one_active_per_slot
    ON bookings(slot_id) WHERE status = 'active';
CREATE UNIQUE INDEX IF NOT EXISTS one_active_per_vehicle
    ON bookings(vehicle) WHERE status = 'active';

app.py

"""ParkBay - parking management backend (Flask + SQLite).

DBMS ............ SQLite tables + unique indexes (schema.sql)
Data structures . min-heap (nearest free slot), deque (FIFO waiting list),
                  dict (slot id -> booking lookup)
"""
import heapq
import os
import re
import sqlite3
from collections import deque
from datetime import datetime

from flask import Flask, g, jsonify, render_template, request

BASE = os.path.dirname(os.path.abspath(__file__))
DB = os.path.join(BASE, "parking.db")
RATE = 20                       # rupees per hour
BLOCKS = {"A": 12, "B": 12}     # block -> number of slots
PLATE = re.compile(r"^[A-Z0-9]{6,12}$")

app = Flask(__name__)
waitlist = deque()              # FIFO queue of dicts: name, vehicle, hours


# ---------- database helpers ----------
def get_db():
    if "db" not in g:
        g.db = sqlite3.connect(DB, timeout=10, isolation_level=None)
        g.db.row_factory = sqlite3.Row
    return g.db


@app.teardown_appcontext
def close_db(_):
    con = g.pop("db", None)
    if con:
        con.close()


def init_db():
    con = sqlite3.connect(DB)
    with open(os.path.join(BASE, "schema.sql")) as f:
        con.executescript(f.read())
    if con.execute("SELECT COUNT(*) FROM slots").fetchone()[0] == 0:
        for block, n in BLOCKS.items():
            for i in range(1, n + 1):
                con.execute("INSERT INTO slots VALUES (?,?,?)", (f"{block}-{i:02d}", block, i))
        con.commit()
    con.close()


def now():
    return datetime.now().isoformat(timespec="seconds")


# ---------- data-structure helpers ----------
def nearest_free_slot(db):
    """Min-heap ordered by (block, number): pops the nearest free slot."""
    rows = db.execute(
        "SELECT block, number, id FROM slots "
        "WHERE id NOT IN (SELECT slot_id FROM bookings WHERE status='active')"
    ).fetchall()
    heap = [(r["block"], r["number"], r["id"]) for r in rows]
    heapq.heapify(heap)
    return heapq.heappop(heap)[2] if heap else None


def create_booking(db, slot, name, vehicle, hours):
    """Insert a booking. Returns an error string, or None on success."""
    try:
        db.execute("BEGIN IMMEDIATE")
        db.execute(
            "INSERT INTO bookings(slot_id,name,vehicle,hours,time_in,status) "
            "VALUES (?,?,?,?,?, 'active')", (slot, name, vehicle, hours, now()))
        db.execute("COMMIT")
        return None
    except sqlite3.IntegrityError as e:
        db.execute("ROLLBACK")
        if "vehicle" in str(e):
            return "This vehicle is already parked."
        return f"Slot {slot} was just taken. Pick another."


def read_person(data):
    name = str(data.get("name", "")).strip()[:30]
    vehicle = re.sub(r"\s+", "", str(data.get("vehicle", ""))).upper()
    try:
        hours = int(data.get("hours", 0))
    except (TypeError, ValueError):
        hours = 0
    if not name:
        return None, "Enter the driver name."
    if not PLATE.match(vehicle):
        return None, "Enter a valid vehicle number, like MH02AB1234."
    if not 1 <= hours <= 12:
        return None, "Duration must be 1 to 12 hours."
    return (name, vehicle, hours), None


def err(msg, code=400):
    return jsonify(error=msg), code


# ---------- routes ----------
@app.get("/")
def home():
    return render_template("index.html", rate=RATE)


@app.get("/api/slots")
def api_slots():
    db = get_db()
    rows = db.execute(
        "SELECT s.id, s.block, b.name, b.vehicle, b.hours, b.time_in "
        "FROM slots s LEFT JOIN bookings b ON b.slot_id = s.id AND b.status='active' "
        "ORDER BY s.block, s.number").fetchall()
    slots = [dict(id=r["id"], block=r["block"], busy=r["vehicle"] is not None,
                  name=r["name"], vehicle=r["vehicle"], hours=r["hours"],
                  time_in=r["time_in"]) for r in rows]
    return jsonify(slots=slots, waiting=len(waitlist), rate=RATE)


@app.post("/api/book")
def api_book():
    data = request.get_json(silent=True) or {}
    person, e = read_person(data)
    if e:
        return err(e)
    db = get_db()
    slot = data.get("slot") or nearest_free_slot(db)      # no slot chosen -> heap picks one
    if not slot:
        return err("The lot is full. You can join the waiting list.", 409)
    if not db.execute("SELECT 1 FROM slots WHERE id=?", (slot,)).fetchone():
        return err("Unknown slot.")
    e = create_booking(db, slot, *person)
    if e:
        return err(e, 409)
    return jsonify(ok=True, slot=slot, fee=person[2] * RATE)


@app.post("/api/release")
def api_release():
    slot = (request.get_json(silent=True) or {}).get("slot")
    db = get_db()
    cur = db.execute(
        "UPDATE bookings SET status='done', time_out=? WHERE slot_id=? AND status='active'",
        (now(), slot))
    if cur.rowcount == 0:
        return err("That slot is not occupied.", 404)
    assigned = None
    while waitlist:                                        # FIFO: first in line gets the slot
        nxt = waitlist.popleft()
        if create_booking(db, slot, nxt["name"], nxt["vehicle"], nxt["hours"]) is None:
            assigned = nxt["name"]
            break
    return jsonify(ok=True, assigned_to=assigned)


@app.post("/api/waitlist")
def api_waitlist():
    data = request.get_json(silent=True) or {}
    person, e = read_person(data)
    if e:
        return err(e)
    db = get_db()
    if nearest_free_slot(db):
        return err("Slots are available. Book one directly.", 409)
    if any(w["vehicle"] == person[1] for w in waitlist):
        return err("This vehicle is already on the waiting list.", 409)
    waitlist.append(dict(name=person[0], vehicle=person[1], hours=person[2]))
    return jsonify(ok=True, position=len(waitlist))


if __name__ == "__main__":
    init_db()
    app.run(debug=True)

var RATE={{ rate }};
var data={slots:[],waiting:0};
var picked=null;
function $(id){return document.getElementById(id);}
function esc(s){var d=document.createElement("div");d.textContent=s;return d.innerHTML;}
function setMsg(t,c){var m=$("msg");m.textContent=t;m.className="msg "+(c||"");}
async function api(url,body){
 try{var r=await fetch(url,body?{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify(body)}:undefined);
  var j=await r.json();if(!r.ok)throw new Error(j.error||"Request failed");return j;}
 catch(e){throw new Error(e.message==="Failed to fetch"?"Cannot reach the server. Is app.py running?":e.message);}
}
function form(){return {name:$("name").value,vehicle:$("veh").value,hours:+$("hrs").value};}
async function load(){
 try{data=await api("/api/slots");
  if(picked&&data.slots.some(function(s){return s.id===picked&&s.busy;}))picked=null;
  render();}catch(e){setMsg(e.message,"err");}
}
function renderLot(){
 var h="",blocks={};
 data.slots.forEach(function(s){(blocks[s.block]=blocks[s.block]||[]).push(s);});
 Object.keys(blocks).forEach(function(b){
  var list=blocks[b],half=list.length/2;
  function bays(a){return a.map(function(s){
   var cls="bay"+(s.busy?" busy":"")+(s.id===picked?" pick":"");
   var lab="Slot "+s.id+(s.busy?", occupied":", available"+(s.id===picked?", selected":""));
   return '<button type="button" class="'+cls+'" data-id="'+s.id+'" aria-label="'+lab+'">'+s.id.split("-")[1]+(s.busy?'<small>'+esc(s.vehicle)+'</small>':'')+'</button>';
  }).join("");}
  h+='<div class="block"><h2>Block '+b+'</h2><div class="row">'+bays(list.slice(0,half))+'</div><div class="lane">DRIVE LANE</div><div class="row">'+bays(list.slice(half))+'</div></div>';
 });
 $("lot").innerHTML=h;
}
function render(){
 var busy=data.slots.filter(function(s){return s.busy;});
 var free=data.slots.length-busy.length;
 $("sTotal").textContent=data.slots.length;$("sBusy").textContent=busy.length;$("sFree").textContent=free;

 $("selTxt").textContent=picked||"none yet";

 $("waitBtn").hidden=free>0;

 if(!busy.length){$("tbl").innerHTML="<p>No active bookings. Pick a green bay to make one.</p>";return;}
 var r=busy.map(function(s){

 $("tbl").innerHTML="<table><tr><th>Slot</th><th>Driver</th><th>Vehicle</th><th>In</th><th>Hours</th><th>Fee</th><th></th></tr>"+r+"</table>";
}

 picked=(picked===id)?null:id;setMsg("");render();

};
